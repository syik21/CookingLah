<%@ Page Language="C#" %>
<script runat="server">
protected void Page_Load(object sender, EventArgs e) {
    if (Request.QueryString["status"] != null) {
        using (var connection = CookingLah.F_Services.DatabaseManager.ConnectDB())
        using (var command = new System.Data.SqlClient.SqlCommand("UPDATE Courses SET Status=@status WHERE CourseID=1", connection)) {
            command.Parameters.AddWithValue("@status", Request.QueryString["status"]);
            connection.Open(); command.ExecuteNonQuery();
        }
        Response.Write("changed"); return;
    }
    if (Request.QueryString["inspect"] != null) {
        var result = new System.Collections.Generic.Dictionary<string,object>();
        using (var connection = CookingLah.F_Services.DatabaseManager.ConnectDB()) {
            connection.Open();
            foreach (string table in new[] { "Courses","Lessons","CourseRecipes","Quizzes","QuizQuestions","QuizOptions","QuizAttempts","PracticalActivities","ActivitySubmissions","CourseProgress","Reviews","CourseReports" }) {
                using(var command = new System.Data.SqlClient.SqlCommand("SELECT COUNT(*) FROM "+table, connection))
                    result[table] = command.ExecuteScalar();
            }
            using(var command = new System.Data.SqlClient.SqlCommand("SELECT LessonID,LessonOrder FROM Lessons WHERE CourseID=1 ORDER BY LessonOrder", connection))
            using(var reader=command.ExecuteReader()) {
                var order = new System.Collections.Generic.List<string>();
                while(reader.Read()) order.Add(reader.GetInt32(0)+":"+reader.GetInt32(1));
                result["LessonOrder"]=order;
            }
            using(var command = new System.Data.SqlClient.SqlCommand("SELECT MIN(RecipeID) FROM CourseRecipes WHERE CourseID=1", connection)) result["RecipeID"]=command.ExecuteScalar();
            using(var command = new System.Data.SqlClient.SqlCommand("SELECT MIN(QuizID) FROM Quizzes WHERE CourseID=1", connection)) result["QuizID"]=command.ExecuteScalar();
            using(var command = new System.Data.SqlClient.SqlCommand("SELECT MIN(ActivityID) FROM PracticalActivities WHERE CourseID=1", connection)) result["ActivityID"]=command.ExecuteScalar();
        }
        Response.ContentType="application/json";
        Response.Write(new System.Web.Script.Serialization.JavaScriptSerializer().Serialize(result)); return;
    }
    Session.Clear();
    Session["UserId"] = Request.QueryString["learner"]=="1" ? 3 : 1;
    Session["Role"] = Request.QueryString["learner"]=="1" ? "Learner" : "Instructor";
    Session["Username"] = "Test Chef";
    Session["Email"] = "chef1@example.invalid";
    Response.Write("fixture ready");
}
</script>
